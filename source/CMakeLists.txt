cmake_minimum_required(VERSION 3.22)
project(CaptionsEverywhere VERSION 1.4.2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PULSE REQUIRED IMPORTED_TARGET libpulse)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/whisper.cpp EXCLUDE_FROM_ALL)
add_library(sherpa SHARED IMPORTED)
set_target_properties(sherpa PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/vendor/sherpa/lib/libsherpa-onnx-c-api.so")
add_executable(captions-everywhere src/main.cpp src/i18n.cpp src/desktop.cpp src/audio.cpp src/recognizer.cpp src/output.cpp src/window.cpp)
target_include_directories(captions-everywhere PRIVATE src vendor/sherpa)
target_link_libraries(captions-everywhere PRIVATE whisper sherpa Qt6::Widgets Qt6::Network PkgConfig::PULSE)
set_target_properties(captions-everywhere PROPERTIES BUILD_RPATH "$ORIGIN/../lib" INSTALL_RPATH "$ORIGIN/../lib")
install(TARGETS captions-everywhere RUNTIME DESTINATION bin)
option(CE_BUILD_CHECKS "Build short UI, persistence and transcript checks" OFF)
if(CE_BUILD_CHECKS)
    add_executable(captions-feature-checks tests/features.cpp src/i18n.cpp src/desktop.cpp src/audio.cpp src/recognizer.cpp src/output.cpp src/window.cpp)
    target_include_directories(captions-feature-checks PRIVATE src vendor/sherpa)
    target_link_libraries(captions-feature-checks PRIVATE whisper sherpa Qt6::Widgets Qt6::Network PkgConfig::PULSE)
    set_target_properties(captions-feature-checks PROPERTIES BUILD_RPATH "$ORIGIN/../lib")
endif()
